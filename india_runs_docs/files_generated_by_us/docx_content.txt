1.2  Recommended Build Order (Why This Order)
The build order is dependency-driven. Each stage depends on the previous one being correct before it is worth optimizing the next:

Environment Setup — nothing works without this.
Stage 1 (Hard Filter) first — cheapest to write, kills ~40% of candidates before any scoring runs. You get immediate runtime savings.
Stage 2B (Skill Trust Score) second — highest signal-per-schema-field ratio. Directly uses skills[], skill_assessment_scores, endorsements.
Stage 2C (Embedding Similarity) third — precomputed, O(1) at ranking time. Needs Stage 0 index built first.
Stage 0 (Offline Precompute) — built before Stage 2C but run once offline. Listed here in coding order.
Stage 2A (Title + Career) — uses TF-IDF and career[]. Moderate complexity.
Stage 3 (Availability Multiplier) — simple additive logic on redrob_signals fields.
Stage 4 (Sort + CSV) — wire everything together, produce output.
Full Pipeline Runner + Validation — end-to-end test.

3.2  Candidate Text Representation
Each candidate is represented as a single concatenated string built from their most informative structured fields. This string is fed to the sentence transformer for embedding.


3.3  Code Blueprint: precompute.py
import json, pickle, numpy as np
from sentence_transformers import SentenceTransformer
from sklearn.feature_extraction.text import TfidfVectorizer
import faiss
from pathlib import Path
from tqdm import tqdm

MODEL_NAME = 'sentence-transformers/all-MiniLM-L6-v2'  # 384-dim, fast
OUT_DIR    = Path('data/precomputed')
OUT_DIR.mkdir(parents=True, exist_ok=True)

def build_cand_text(c: dict) -> str:
    p = c['profile']
    skills = ' '.join(s['name'] for s in c.get('skills', []))
    career_text = ' '.join(
        (h['title'] + ' ' + h['description'][:500])
        for h in c.get('career_history', [])
    )
    return f"{p['current_title']} {p.get('headline','')} {skills} {career_text}"

def build_jd_text(jd: dict) -> str:
    skills = ' '.join(jd['must_have_skills'] + jd.get('nice_to_have_skills', []))
    return f"{jd['title']} {skills} {jd.get('description','')}"

def run_precompute(candidates_path, jd_path):
    candidates = json.loads(Path(candidates_path).read_text())
    jd         = json.loads(Path(jd_path).read_text())

    model = SentenceTransformer(MODEL_NAME)

    # 1. Build candidate texts
    cand_ids   = [c['candidate_id'] for c in candidates]
    cand_texts = [build_cand_text(c) for c in tqdm(candidates, desc='Building texts')]

    # 2. Embed all candidates (batch=256 for memory efficiency)
    print('Embedding candidates ...')
    cand_vecs = model.encode(
        cand_texts, batch_size=256,
        normalize_embeddings=True,   # cosine = dot product on L2-norm vecs
        show_progress_bar=True
    ).astype('float32')

    # 3. Embed JD
    jd_vec = model.encode([build_jd_text(jd)],
                          normalize_embeddings=True).astype('float32')

    # 4. Build FAISS index (flat = exact, no approximation)
    index = faiss.IndexFlatIP(cand_vecs.shape[1])  # inner product = cosine
    index.add(cand_vecs)

    # 5. Fit TF-IDF on candidate corpus
    tfidf = TfidfVectorizer(max_features=30000, ngram_range=(1,2))
    tfidf.fit(cand_texts)

    # 6. Save all artifacts
    np.save(OUT_DIR / 'jd_vec.npy', jd_vec)
    np.save(OUT_DIR / 'cand_vecs.npy', cand_vecs)
    (OUT_DIR / 'cand_ids.json').write_text(json.dumps(cand_ids))
    faiss.write_index(index, str(OUT_DIR / 'faiss.index'))
    with open(OUT_DIR / 'tfidf.pkl', 'wb') as f: pickle.dump(tfidf, f)

    print(f'Done. {len(cand_ids)} candidates indexed.')

if __name__ == '__main__':
    run_precompute('data/candidates.json', 'data/job_description.json')

4.3  Code Blueprint: hard_filter.py
from datetime import datetime, timezone
from sklearn.metrics.pairwise import cosine_similarity
import pickle, numpy as np

def load_tfidf(path='data/precomputed/tfidf.pkl'):
    with open(path, 'rb') as f: return pickle.load(f)

def is_killed(candidate: dict, jd: dict, tfidf, today=None) -> tuple[bool, str]:
    """Returns (killed: bool, reason: str)."""
    if today is None:
        today = datetime.now(timezone.utc).date()

    sig = candidate['redrob_signals']
    prof = candidate['profile']

    # Rule 1: ghost profile
    if sig['profile_completeness_score'] < 30:
        return True, 'profile_completeness < 30'

    # Rule 2: unverified identity
    if not sig['verified_email']:
        return True, 'unverified_email'

    # Rule 3: serial no-show
    if sig['interview_completion_rate'] < 0.20:
        return True, 'interview_completion_rate < 0.20'

    # Rule 4: stale profile
    from dateutil.parser import parse as parse_date
    last_active = parse_date(sig['last_active_date']).date()
    if (today - last_active).days > 180:
        return True, 'inactive > 180 days'

    # Rule 5: not open to work
    if not sig['open_to_work_flag']:
        return True, 'not_open_to_work'

    # Rule 6: zero industry overlap
    cand_industries = {prof.get('current_industry', '')}
    cand_industries |= {h['industry'] for h in candidate.get('career_history', [])}
    jd_industries = set(jd.get('target_industries', []))
    if jd_industries and len(cand_industries & jd_industries) == 0:
        return True, 'no_industry_overlap'

    # Rule 7: title completely off-topic
    cand_title_vec = tfidf.transform([prof['current_title']])
    jd_title_vec   = tfidf.transform([jd['title']])
    sim = cosine_similarity(cand_title_vec, jd_title_vec)[0, 0]
    if sim < 0.05:
        return True, f'title_similarity={sim:.3f}<0.05'

    # Rule 8: honeypot detection
    gh = sig.get('github_activity_score', -1)
    no_assessment = len(sig.get('skill_assessment_scores', {})) == 0
    no_endorse    = sig.get('endorsements_received', 0) == 0
    no_connections= sig.get('connection_count', 0) < 5
    if gh == -1 and no_assessment and no_endorse and no_connections:
        return True, 'honeypot_ghost_profile'

    return False, ''

def apply_hard_filter(candidates, jd, tfidf):
    """Returns (survivors, killed_ids)."""
    survivors, killed = [], []
    for c in candidates:
        killed_flag, reason = is_killed(c, jd, tfidf)
        if killed_flag:
            killed.append({'candidate_id': c['candidate_id'], 'reason': reason})
        else:
            survivors.append(c)
    print(f'Hard filter: {len(killed)} killed, {len(survivors)} survivors')
    return survivors, killed

5.3  Code Blueprint: score_career.py
import numpy as np, pickle
from datetime import datetime, timezone
from dateutil.parser import parse as parse_date
from sklearn.metrics.pairwise import cosine_similarity
import re

DECAY_LAMBDA = 0.15   # half-life ~ 4.6 years

def compute_A(candidate: dict, jd: dict, tfidf, keyword_max: float) -> dict:
    prof = candidate['profile']
    today = datetime.now(timezone.utc).date()

    # ── title_sim ──────────────────────────────────────
    cand_title_vec = tfidf.transform([prof['current_title']])
    jd_title_vec   = tfidf.transform([jd['title']])
    title_sim = float(cosine_similarity(cand_title_vec, jd_title_vec)[0, 0])

    # ── industry_match ────────────────────────────────
    cand_inds = {prof.get('current_industry', '')}
    cand_inds |= {h['industry'] for h in candidate.get('career_history', [])}
    jd_inds = set(jd.get('target_industries', []))
    matched = len(cand_inds & jd_inds) if jd_inds else 0
    industry_match = min(matched / max(len(jd_inds), 1), 1.0)

    # ── prod_keyword_density (recency decayed) ────────
    keywords = [kw.lower() for kw in jd.get('keywords', [])]
    weighted_hits = 0.0
    for h in candidate.get('career_history', []):
        if h.get('is_current', False):
            age_yrs = 0.0
        else:
            start = parse_date(h['start_date']).date()
            age_yrs = max((today - start).days / 365, 0)
        decay = np.exp(-DECAY_LAMBDA * age_yrs)
        desc = h.get('description', '').lower()
        hits = sum(1 for kw in keywords if kw in desc)
        weighted_hits += hits * decay
    # normalize against pool max (passed in from pipeline)
    prod_keyword_density = weighted_hits / max(keyword_max, 1e-9)
    prod_keyword_density = min(prod_keyword_density, 1.0)

    # ── yoe_score ─────────────────────────────────────
    yoe = prof.get('years_of_experience', 0)
    min_yoe = max(jd.get('min_yoe', 1), 1)
    yoe_score = min(yoe / min_yoe, 1.0)

    A = (0.35 * title_sim
       + 0.25 * industry_match
       + 0.25 * prod_keyword_density
       + 0.15 * yoe_score)

    return {
        'A': round(A, 4),
        'title_sim': round(title_sim, 4),
        'industry_match': round(industry_match, 4),
        'prod_keyword_density': round(prod_keyword_density, 4),
        'yoe_score': round(yoe_score, 4),
    }

def compute_keyword_max(candidates, jd, tfidf):
    """Pre-compute the max weighted hits across the pool for normalization."""
    keywords = [kw.lower() for kw in jd.get('keywords', [])]
    today = datetime.now(timezone.utc).date()
    max_hits = 0.0
    for c in candidates:
        wh = 0.0
        for h in c.get('career_history', []):
            age_yrs = 0.0 if h.get('is_current') else \
                      max((today - parse_date(h['start_date']).date()).days/365, 0)
            decay = np.exp(-DECAY_LAMBDA * age_yrs)
            desc = h.get('description','').lower()
            wh += sum(1 for kw in keywords if kw in desc) * decay
        max_hits = max(max_hits, wh)
    return max_hits

from rapidfuzz import fuzz

PROF_WEIGHT = {'beginner': 0.40, 'intermediate': 0.70,
               'advanced': 0.90, 'expert': 1.00}

def fuzzy_match_skill(cand_skill_name: str, jd_skills: list[str]) -> str | None:
    """Returns the matching JD skill name if fuzzy score >= 85, else None."""
    best_score, best_match = 0, None
    for jd_skill in jd_skills:
        score = fuzz.token_sort_ratio(cand_skill_name.lower(), jd_skill.lower())
        if score > best_score:
            best_score, best_match = score, jd_skill
    return best_match if best_score >= 85 else None

def skill_trust(skill_obj: dict, assess_scores: dict, jd_skill: str) -> float:
    """Compute trust score for a single matched skill."""
    prof_w    = PROF_WEIGHT.get(skill_obj.get('proficiency','beginner'), 0.40)
    endorse_w = min(skill_obj.get('endorsements', 0) / 20.0, 1.0)
    assess_key = next((k for k in assess_scores if jd_skill.lower() in k.lower()), None)
    assess_w  = (assess_scores[assess_key] / 100.0) if assess_key else 0.0
    duration_w= min(skill_obj.get('duration_months', 0) / 24.0, 1.0)
    return prof_w * (0.35 + 0.25*endorse_w + 0.25*assess_w + 0.15*duration_w)

def compute_B(candidate: dict, jd: dict) -> dict:
    sig = candidate['redrob_signals']
    assess_scores = sig.get('skill_assessment_scores', {})
    cand_skills = {s['name']: s for s in candidate.get('skills', [])}
    must_have = jd.get('must_have_skills', [])
    nice_have = jd.get('nice_to_have_skills', [])

    def coverage(jd_skills: list) -> float:
        if not jd_skills: return 0.0
        total = 0.0
        for jd_skill in jd_skills:
            matched_name = fuzzy_match_skill(jd_skill, list(cand_skills.keys()))
            if matched_name:
                total += skill_trust(cand_skills[matched_name], assess_scores, jd_skill)
        return total / len(jd_skills)

    must_cov = coverage(must_have)
    nice_cov = coverage(nice_have)

    # Certification bonus
    all_jd_skills = must_have + nice_have
    cert_bonus = 0.0
    for cert in candidate.get('certifications', []):
        if fuzzy_match_skill(cert['name'], all_jd_skills) is not None:
            cert_bonus = min(cert_bonus + 0.05, 0.15)

    B = min(0.75 * must_cov + 0.25 * nice_cov + cert_bonus, 1.0)
    return {
        'B': round(B, 4),
        'must_have_coverage': round(must_cov, 4),
        'nice_coverage': round(nice_cov, 4),
        'cert_bonus': round(cert_bonus, 4),
    }

import numpy as np, json, faiss
from pathlib import Path

def load_artifacts(precomputed_dir='data/precomputed'):
    d = Path(precomputed_dir)
    jd_vec    = np.load(d / 'jd_vec.npy')         # shape: [1, 384]
    cand_vecs = np.load(d / 'cand_vecs.npy')      # shape: [N, 384]
    cand_ids  = json.loads((d / 'cand_ids.json').read_text())
    return jd_vec, cand_vecs, cand_ids

def compute_C_all(jd_vec, cand_vecs) -> np.ndarray:
    """Returns similarity scores for all candidates. Shape: [N,]."""
    # Both are already L2-normalized → dot = cosine
    scores = (cand_vecs @ jd_vec.T).squeeze()  # [N,]
    scores = np.clip(scores, 0.0, 1.0)         # guard against float errors
    return scores

def get_C_map(jd_vec, cand_vecs, cand_ids) -> dict:
    """Returns {candidate_id: C_score} dict."""
    scores = compute_C_all(jd_vec, cand_vecs)
    return {cid: float(scores[i]) for i, cid in enumerate(cand_ids)}

from src.score_career import compute_A, compute_keyword_max
from src.score_skills import compute_B
from src.score_embed import load_artifacts, get_C_map
import pickle

def compute_raw_scores(survivors: list, jd: dict) -> list:
    """
    Returns list of dicts with candidate_id, A, B, C, raw_score,
    plus component sub-scores for reasoning generation.
    """
    # Load precomputed artifacts once
    with open('data/precomputed/tfidf.pkl', 'rb') as f:
        tfidf = pickle.load(f)
    jd_vec, cand_vecs, cand_ids = load_artifacts()
    C_map = get_C_map(jd_vec, cand_vecs, cand_ids)

    # Pre-compute keyword max for normalization
    keyword_max = compute_keyword_max(survivors, jd, tfidf)

    results = []
    for c in survivors:
        cid = c['candidate_id']
        a_result = compute_A(c, jd, tfidf, keyword_max)
        b_result = compute_B(c, jd)
        C = C_map.get(cid, 0.0)

        A = a_result['A']
        B = b_result['B']
        raw = round(0.40*A + 0.35*B + 0.25*C, 4)

        results.append({
            'candidate_id': cid,
            'candidate': c,
            'A': A, 'B': B, 'C': round(C, 4),
            'raw_score': raw,
            **{k: v for k, v in a_result.items() if k != 'A'},
            **{k: v for k, v in b_result.items() if k != 'B'},
        })

    return results

from datetime import datetime, timezone
from dateutil.parser import parse as parse_date

def compute_multipliers(candidate: dict, jd: dict, today=None) -> dict:
    if today is None:
        today = datetime.now(timezone.utc).date()
    sig  = candidate['redrob_signals']
    prof = candidate['profile']

    # ── Availability multiplier ─────────────────────────
    a_mult = 1.0
    last_active = parse_date(sig['last_active_date']).date()
    days_since  = (today - last_active).days

    if sig.get('open_to_work_flag', False):           a_mult += 0.10
    if days_since <= 14:                              a_mult += 0.10
    if days_since <= 7:                               a_mult += 0.05  # stacks
    if sig.get('recruiter_response_rate', 0) >= 0.70: a_mult += 0.05
    oar = sig.get('offer_acceptance_rate', -1)
    if oar != -1 and oar >= 0.80:                     a_mult += 0.05
    if sig.get('avg_response_time_hours', 0) > 72:    a_mult -= 0.05
    if sig.get('notice_period_days', 0) > 90:         a_mult -= 0.10
    if sig.get('interview_completion_rate', 1) < 0.50:a_mult -= 0.15
    if days_since > 90:                               a_mult -= 0.25

    budget_max = jd.get('budget_max_inr_lpa', None)
    sal = sig.get('expected_salary_range_inr_lpa', {})
    sal_min = sal.get('min', 0)
    if budget_max and sal_min > budget_max:            a_mult -= 0.20

    a_mult = max(0.50, min(a_mult, 1.25))

    # ── Location multiplier ─────────────────────────────
    preferred = set(l.lower() for l in jd.get('preferred_locations', []))
    cand_loc  = prof.get('location', '').lower()
    relocate  = sig.get('willing_to_relocate', False)
    l_mult = 1.0
    if preferred and any(p in cand_loc for p in preferred):  l_mult += 0.05
    elif preferred and not relocate:                          l_mult -= 0.05
    l_mult = max(0.70, min(l_mult, 1.05))

    return {'availability_mult': round(a_mult, 4),
            'location_mult': round(l_mult, 4)}

def apply_multipliers(scored_results: list, jd: dict) -> list:
    for r in scored_results:
        mults = compute_multipliers(r['candidate'], jd)
        r.update(mults)
        r['final_score'] = round(
            min(r['raw_score'] * mults['availability_mult'] * mults['location_mult'], 1.0),
            4
        )
    return scored_results

import pandas as pd

def generate_reasoning(r: dict, jd: dict) -> str:
    c   = r['candidate']
    sig = c['redrob_signals']
    prof= c['profile']
    cov = r.get('must_have_coverage', 0)

    gh_note = (f"GitHub score: {sig['github_activity_score']}"
               if sig.get('github_activity_score', -1) != -1 else 'No GitHub linked')

        f"{prof['current_title']} @ {prof['current_company']} | "

        f"Notice: {sig['notice_period_days']}d | "
        f"{gh_note} | "
        f"Score: {r['final_score']:.4f} "
        f"(A={r['A']:.3f} B={r['B']:.3f} C={r['C']:.3f})"

def write_submission(results: list, jd: dict, out_path='submission.csv'):
    # Sort by final_score DESC, tiebreak by profile_completeness, saved_by_recruiters
    def sort_key(r):
        sig = r['candidate']['redrob_signals']
        return (
            -r['final_score'],
            -sig.get('profile_completeness_score', 0),
            -sig.get('saved_by_recruiters_30d', 0),
        )
    results.sort(key=sort_key)
    top100 = results[:100]

    rows = []
    for rank, r in enumerate(top100, start=1):
        rows.append({
            'rank':          rank,
            'candidate_id':  r['candidate_id'],
            'final_score':   r['final_score'],
            'raw_score':     r['raw_score'],
            'component_scores': f"A={r['A']:.3f}|B={r['B']:.3f}|C={r['C']:.3f}",
            'availability_mult': r['availability_mult'],
            'location_mult': r['location_mult'],
            'reasoning':     generate_reasoning(r, jd),
        })

    df = pd.DataFrame(rows)
    df.to_csv(out_path, index=False)
    print(f'Submission written: {out_path} ({len(rows)} candidates)')
    return df

  For each career role k:
    age_k = (today - career[k].start_date).days / 365
    if career[k].is_current: age_k = 0

    weighted_hits += hits_k × decay_k
  prod_keyword_density = weighted_hits / max(weighted_hits_pool)
